    %% Inheritance-like relations
    MESSAGE ||--o{ POST : "is subtype"
    MESSAGE ||--o{ COMMENT : "is subtype"
    PLACE ||--o{ PLACE : "isPartOf"
    ORGANISATION }o--|| PLACE : "isLocatedIn"

    %% Person relations
    PERSON }o--o{ PERSON : "knows"
    PERSON }o--o{ TAG : "hasInterest"
    PERSON }o--|| PLACE : "isLocatedIn"
    PERSON }o--o{ ORGANISATION : "studyAt"
    PERSON }o--o{ ORGANISATION : "workAt"

    %% Forum relations
    FORUM ||--o{ POST : "containerOf"
    FORUM }o--|| PERSON : "hasModerator"
    FORUM }o--o{ PERSON : "hasMember"
    FORUM }o--o{ TAG : "hasTag"

    %% Message relations
    POST }o--|| PERSON : "hasCreator"
    COMMENT }o--|| PERSON : "hasCreator"
    POST }o--|| PLACE : "isLocatedIn"
    COMMENT }o--|| PLACE : "isLocatedIn"

    POST }o--o{ TAG : "hasTag"
    COMMENT }o--o{ TAG : "hasTag"

    PERSON }o--o{ POST : "likes"
    PERSON }o--o{ COMMENT : "likes"

    COMMENT }o--|| POST : "replyOf"
    COMMENT }o--|| COMMENT : "replyOf"

    %% Tag hierarchy
    TAG }o--|| TAGCLASS : "hasType"
    TAGCLASS }o--o| TAGCLASS : "isSubclassOf"
